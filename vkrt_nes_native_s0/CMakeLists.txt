# Windows facilities belong to the platform link set; the only product library
# dependency remains vkrt_nes_m0. This also preserves the historical static gate.
if(WIN32)
  if(MSVC)
    set(CMAKE_CXX_STANDARD_LIBRARIES "${CMAKE_CXX_STANDARD_LIBRARIES} user32.lib gdi32.lib shell32.lib comdlg32.lib ole32.lib uuid.lib winmm.lib ksuser.lib d3d11.lib dxgi.lib")
  else()
    set(CMAKE_CXX_STANDARD_LIBRARIES "${CMAKE_CXX_STANDARD_LIBRARIES} -luser32 -lgdi32 -lshell32 -lcomdlg32 -lole32 -luuid -lwinmm -lksuser -ld3d11 -ldxgi")
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -static -static-libgcc -static-libstdc++")
  endif()
  set(VKRT_FRONTEND_SOURCES app/windows/frontend_config.cpp
    app/windows/frontend_input.cpp app/windows/frontend_ui.cpp)
  add_executable(vkrt_nes_runtime WIN32 app/windows/runtime_main.cpp
    app/windows/runtime_audio.cpp app/windows/runtime_presenter.cpp app/windows/frontend_product.cpp ${VKRT_FRONTEND_SOURCES})
  target_include_directories(vkrt_nes_runtime PRIVATE app/windows)
  target_link_libraries(vkrt_nes_runtime PRIVATE vkrt_nes_m0)
  add_executable(vkrt_nes_m5_ui_tests tests/test_m5_ui.cpp ${VKRT_FRONTEND_SOURCES})
  target_include_directories(vkrt_nes_m5_ui_tests PRIVATE app/windows)
  target_link_libraries(vkrt_nes_m5_ui_tests PRIVATE vkrt_nes_m0)
  add_executable(vkrt_nes_m5r1_product_tests tests/test_m5r1_product.cpp
    app/windows/frontend_product.cpp app/windows/runtime_audio.cpp ${VKRT_FRONTEND_SOURCES})
  target_include_directories(vkrt_nes_m5r1_product_tests PRIVATE app/windows)
  target_link_libraries(vkrt_nes_m5r1_product_tests PRIVATE vkrt_nes_m0)
  add_test(NAME vkrt_nes_m5r1_product_tests COMMAND vkrt_nes_m5r1_product_tests)
  add_executable(vkrt_nes_m5r11_polish_tests tests/test_m5r11_polish.cpp ${VKRT_FRONTEND_SOURCES})
  target_include_directories(vkrt_nes_m5r11_polish_tests PRIVATE app/windows)
  target_link_libraries(vkrt_nes_m5r11_polish_tests PRIVATE vkrt_nes_m0)
